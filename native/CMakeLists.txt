cmake_minimum_required(VERSION 3.22)
project(StockfishAndroid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SF ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/stockfish/src)
file(GLOB ENGINE_SOURCES ${SF}/*.cpp ${SF}/nnue/*.cpp ${SF}/nnue/features/*.cpp ${SF}/syzygy/*.cpp)
add_executable(stockfish ${ENGINE_SOURCES})
target_compile_definitions(stockfish PRIVATE NDEBUG IS_64BIT USE_NEON USE_POPCNT NNUE_EMBEDDING_OFF)
target_compile_options(stockfish PRIVATE -O3 -fno-exceptions -fno-rtti -fconstexpr-steps=500000000)
target_link_options(stockfish PRIVATE -Wl,-z,max-page-size=16384)
set_target_properties(stockfish PROPERTIES OUTPUT_NAME libstockfish.so RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../app/src/main/jniLibs/arm64-v8a)
